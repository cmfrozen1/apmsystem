  <script>
    let currentCoordinates = '';
    let userId = '';

    // Initialize LIFF
    async function initializeLiff() {
      try {
        await liff.init({ liffId: '1654032238-V3JmdyRg' }); // Replace with your LIFF ID
        if (liff.isLoggedIn()) {
          const profile = await liff.getProfile();
          userId = profile.userId;
          document.getElementById('profileImage').src = profile.pictureUrl;
          document.getElementById('displayName').textContent = profile.displayName;
        } else {
          liff.login();
        }
      } catch (error) {
        Swal.fire({
          icon: 'error',
          title: 'LIFF Initialization Failed',
          text: error.message
        });
      }
    }

    function getLocation() {
      if (navigator.geolocation) {
        navigator.geolocation.getCurrentPosition(showPosition, showError);
      } else {
        Swal.fire({
          icon: 'error',
          title: 'ไม่สามารถรับพิกัดได้',
          text: 'เบราว์เซอร์ของคุณไม่สนับสนุนการรับพิกัด'
        });
      }
    }

    function showPosition(position) {
      currentCoordinates = position.coords.latitude + ", " + position.coords.longitude;
      const mapHtml = `<iframe width="100%" height="300" frameborder="0" scrolling="no" marginheight="0" marginwidth="0" src="https://maps.google.com/maps?q=${currentCoordinates}&t=&z=15&ie=UTF8&iwloc=&output=embed"></iframe>`;
      document.getElementById('mapContainer').innerHTML = mapHtml;
    }

    function showError(error) {
      Swal.fire({
        icon: 'error',
        title: 'ข้อผิดพลาดตำแหน่ง',
        text: 'ไม่สามารถรับตำแหน่งได้: ' + error.message
      });
    }

    document.getElementById('imageFile').addEventListener('change', function(event) {
      const file = event.target.files[0];
      if (file) {
        const reader = new FileReader();
        reader.onload = function(e) {
          document.getElementById('imagePreview').innerHTML = `<img src="${e.target.result}" alt="Image Preview">`;
        };
        reader.readAsDataURL(file);
      }
    });

    document.getElementById('dataForm').addEventListener('submit', async function(event) {
      event.preventDefault();
      
      const loadingOverlay = document.getElementById('loadingOverlay');
      loadingOverlay.style.display = 'block';

      const submitButton = document.getElementById('sub');
      submitButton.disabled = true;
      submitButton.value = "Submitting..";

      const formData = new FormData(this);
      const fileInput = document.getElementById('imageFile').files[0];
      const submitDate = new Date().toLocaleDateString();
      const submitTime = new Date().toLocaleTimeString();

      const reader = new FileReader();
      reader.onload = async function(event) {
        const base64 = event.target.result.split(',')[1];
        const data = {
          date: formData.get('date'),
          firstName: formData.get('firstName'),
          district: formData.get('district'),
          collector: formData.get('collector'),
          contact: formData.get('contact'),
          plantType: formData.get('plantType'),
          other: formData.get('other'),
          details: formData.get('details'),
          coordinates: currentCoordinates,
          location: `https://maps.google.com/maps?q=${currentCoordinates}`,
          base64: base64,
          type: fileInput.type,
          name: fileInput.name,
          submitDate: submitDate,
          submitTime: submitTime,
          userId: userId
        };

        try {
          // Save to Google Apps Script
          const response = await fetch('https://script.google.com/macros/s/AKfycbzniYO49YaS_3gv-V1WoLakPP14j6D9HScO6XCMDMzl3LdkADSI_RN1gVpaygghwdLD/exec', {
            method: "POST",
            body: JSON.stringify(data)
          });
          const result = await response.text();

          // Create Flex Message
          const flexMessage = {
            type: "flex",
            altText: "รายงานการส่งเสริมการเกษตร",
            contents: {
              type: "bubble",
              body: {
                type: "box",
                layout: "vertical",
                contents: [
                  {
                    type: "text",
                    text: "รายงานการส่งเสริมการเกษตร",
                    weight: "bold",
                    size: "xl"
                  },
                  {
                    type: "box",
                    layout: "vertical",
                    margin: "lg",
                    spacing: "sm",
                    contents: [
                      { type: "text", text: `วันที่: ${data.date}` },
                      { type: "text", text: `ชื่อส่งเสริม: ${data.firstName}` },
                      { type: "text", text: `อำเภอ: ${data.district}` },
                      { type: "text", text: `Collector: ${data.collector}` },
                      { type: "text", text: `ผู้ติดต่อ: ${data.contact}` },
                      { type: "text", text: `ชนิดพืช: ${data.plantType}` },
                      { type: "text", text: `อื่นๆ: ${data.other || '-'}` },
                      { type: "text", text: `รายละเอียด: ${data.details}` },
                      { type: "text", text: `ส่งเมื่อ: ${data.submitDate} ${data.submitTime}` }
                    ]
                  }
                ]
              }
            }
          };

          // Send Flex Message and Location
          await liff.sendMessages([
            flexMessage,
            {
              type: "location",
              title: "สถานที่ทำงาน",
              address: data.district,
              latitude: parseFloat(currentCoordinates.split(',')[0]),
              longitude: parseFloat(currentCoordinates.split(',')[1])
            }
          ]);

          loadingOverlay.style.display = 'none';
          Swal.fire({
            icon: 'success',
            title: 'บันทึกและส่งสำเร็จ!',
            text: 'ข้อมูลถูกบันทึกและส่งไปยัง LINE Chat เรียบร้อยแล้ว',
            timer: 2000,
            showConfirmButton: false
          });
          // Close LIFF window
           liff.closeWindow();

          submitButton.value = "Submit";
          submitButton.disabled = false;
          document.getElementById('dataForm').reset();
          document.getElementById('imagePreview').innerHTML = '';
        } catch (error) {
          loadingOverlay.style.display = 'none';
          Swal.fire({
            icon: 'error',
            title: 'เกิดข้อผิดพลาด',
            text: 'ไม่สามารถบันทึกหรือส่งข้อมูลได้: ' + error.message
          });
          submitButton.value = "Submit";
          submitButton.disabled = false;
        }
      };
      reader.readAsDataURL(fileInput);
    });

    function fetchPromoterNames() {
      fetch('https://script.google.com/macros/s/AKfycbzniYO49YaS_3gv-V1WoLakPP14j6D9HScO6XCMDMzl3LdkADSI_RN1gVpaygghwdLD/exec?action=getPromoterNames')
        .then(response => response.json())
        .then(data => {
          if (data.error) {
            Swal.fire({ icon: 'error', title: 'ข้อผิดพลาด', text: 'ไม่สามารถโหลดชื่อส่งเสริมได้: ' + data.error });
            return;
          }
          const selectElement = $('#firstName');
          data.forEach(name => selectElement.append(new Option(name, name)));
          selectElement.select2({ placeholder: "กรุณาเลือกชื่อส่งเสริม", allowClear: true });
        })
        .catch(error => {
          Swal.fire({ icon: 'error', title: 'ข้อผิดพลาด', text: 'เกิดข้อผิดพลาดในการเชื่อมต่อ: ' + error.message });
        });
    }

    function fetchCollectors() {
      fetch('https://script.google.com/macros/s/AKfycbzniYO49YaS_3gv-V1WoLakPP14j6D9HScO6XCMDMzl3LdkADSI_RN1gVpaygghwdLD/exec?action=getCollectors')
        .then(response => response.json())
        .then(data => {
          if (data.error) {
            Swal.fire({ icon: 'error', title: 'ข้อผิดพลาด', text: 'ไม่สามารถโหลด Collectors ได้: ' + data.error });
            return;
          }
          const selectElement = $('#collector');
          data.forEach(name => selectElement.append(new Option(name, name)));
          selectElement.select2({ placeholder: "กรุณาเลือก Collector", allowClear: true });
        })
        .catch(error => {
          Swal.fire({ icon: 'error', title: 'ข้อผิดพลาด', text: 'เกิดข้อผิดพลาดในการเชื่อมต่อ: ' + error.message });
        });
    }

    function fetchPlantTypes() {
      fetch('https://script.google.com/macros/s/AKfycbzniYO49YaS_3gv-V1WoLakPP14j6D9HScO6XCMDMzl3LdkADSI_RN1gVpaygghwdLD/exec?action=getPlantTypes')
        .then(response => response.json())
        .then(data => {
          if (data.error) {
            Swal.fire({ icon: 'error', title: 'ข้อผิดพลาด', text: 'ไม่สามารถโหลดชนิดพืชได้: ' + data.error });
            return;
          }
          const selectElement = $('#plantType');
          data.forEach(type => selectElement.append(new Option(type, type)));
          selectElement.select2({ placeholder: "กรุณาเลือกชนิดพืช", allowClear: true });
        })
        .catch(error => {
          Swal.fire({ icon: 'error', title: 'ข้อผิดพลาด', text: 'เกิดข้อผิดพลาดในการเชื่อมต่อ: ' + error.message });
        });
    }

    window.onload = function() {
      initializeLiff().then(() => {
        getLocation();
        const today = new Date().toISOString().split('T')[0];
        document.getElementById('date').value = today;
        fetchPromoterNames();
        fetchCollectors();
        fetchPlantTypes();
      });
    };
  </script>

  <script>
    let currentCoordinates = '';
    let userId = '';

    // Initialize LIFF
    async function initializeLiff() {
      try {
        await liff.init({ liffId: '1654032238-V3JmdyRg' }); // Replace with your LIFF ID
        if (liff.isLoggedIn()) {
          const profile = await liff.getProfile();
          userId = profile.userId;
          document.getElementById('profileImage').src = profile.pictureUrl;
          document.getElementById('displayName').textContent = profile.displayName;
        } else {
          liff.login();
        }
      } catch (error) {
        Swal.fire({
          icon: 'error',
          title: 'LIFF Initialization Failed',
          text: error.message
        });
      }
    }

    function getLocation() {
      if (navigator.geolocation) {
        navigator.geolocation.getCurrentPosition(showPosition, showError);
      } else {
        Swal.fire({
          icon: 'error',
          title: 'ไม่สามารถรับพิกัดได้',
          text: 'เบราว์เซอร์ของคุณไม่สนับสนุนการรับพิกัด'
        });
      }
    }

    function showPosition(position) {
      currentCoordinates = position.coords.latitude + ", " + position.coords.longitude;
      const mapHtml = `<iframe width="100%" height="300" frameborder="0" scrolling="no" marginheight="0" marginwidth="0" src="https://maps.google.com/maps?q=${currentCoordinates}&t=&z=15&ie=UTF8&iwloc=&output=embed"></iframe>`;
      document.getElementById('mapContainer').innerHTML = mapHtml;
    }

    function showError(error) {
      Swal.fire({
        icon: 'error',
        title: 'ข้อผิดพลาดตำแหน่ง',
        text: 'ไม่สามารถรับตำแหน่งได้: ' + error.message
      });
    }

    document.getElementById('imageFile').addEventListener('change', function(event) {
      const file = event.target.files[0];
      if (file) {
        const reader = new FileReader();
        reader.onload = function(e) {
          document.getElementById('imagePreview').innerHTML = `<img src="${e.target.result}" alt="Image Preview">`;
        };
        reader.readAsDataURL(file);
      }
    });

    document.getElementById('dataForm').addEventListener('submit', async function(event) {
      event.preventDefault();
      
      const loadingOverlay = document.getElementById('loadingOverlay');
      loadingOverlay.style.display = 'block';

      const submitButton = document.getElementById('sub');
      submitButton.disabled = true;
      submitButton.textContent = "กำลังบันทึก...";

      const formData = new FormData(this);
      const fileInput = document.getElementById('imageFile').files[0];
      const submitDate = new Date().toLocaleDateString();
      const submitTime = new Date().toLocaleTimeString();
      const location = `https://maps.google.com/maps?q=${currentCoordinates}`;

      const reader = new FileReader();
      reader.onload = async function(event) {
        const base64 = event.target.result.split(',')[1];
        const data = {
          date: formData.get('date'),
          firstName: formData.get('firstName'),
          district: formData.get('district'),
          village: formData.get('village'),
          subDistrict: formData.get('subDistrict'),
          details: formData.get('details'),
          coordinates: currentCoordinates,
          location: location,
          base64: base64,
          type: fileInput.type,
          name: fileInput.name,
          submitDate: submitDate,
          submitTime: submitTime,
          userId: userId
        };

        try {
          // Save to Google Apps Script
          const response = await fetch('https://script.google.com/macros/s/AKfycbyE8tvOVYAAR6WOiCVDEb2Fp1Yq3ebTAri9ZQYonpduMjQCFzod8xi0glBVXc6A6_qOeA/exec', {
            method: "POST",
            body: JSON.stringify(data)
          });
          const result = await response.text();

          // Adapted Flex Message
          const flexMessage = {
            type: "flex",
            altText: "รายงานการส่งเสริมการเกษตร",
            contents: {
              type: "bubble",
              body: {
                type: "box",
                layout: "vertical",
                contents: [
                  {
                    type: "text",
                    text: "ส่งงานSuppervisor",
                    weight: "bold",
                    size: "md"
                  },
                  {
                    type: "box",
                    layout: "vertical",
                    margin: "lg",
                    spacing: "sm",
                    contents: [
                      { type: "text", text: `วันที่: ${data.date}` },
                      { type: "text", text: `ชื่อ: ${data.firstName}` },
                      { type: "text", text: `เขต: ${data.district}` },
                      { type: "text", text: `หมู่บ้าน: ${data.village}` },
                      { type: "text", text: `ตำบล: ${data.subDistrict}` },
                      { type: "text", text: `รายละเอียดปฏิบัติงาน: ${data.details}` },
                      { type: "text", text: `พิกัด: ${data.coordinates}` },
                      { type: "text", text: `ส่งเมื่อ: ${data.submitDate} ${data.submitTime}` }
                    ]
                  }
                ]
              }
            }
          };

          // Send Flex Message and Location
          await liff.sendMessages([
            flexMessage,
            {
              type: "location",
              title: "สถานที่ปฏิบัติงาน",
              address: data.location,
              latitude: parseFloat(currentCoordinates.split(',')[0]),
              longitude: parseFloat(currentCoordinates.split(',')[1])
            }
          ]);

          loadingOverlay.style.display = 'none';
          Swal.fire({
            icon: 'success',
            title: 'บันทึกและส่งสำเร็จ!',
            text: 'ข้อมูลการกำกับดูแลถูกบันทึกและส่งไปยัง LINE Chat เรียบร้อยแล้ว',
            timer: 2000,
            showConfirmButton: false
          });
          liff.closeWindow();

          submitButton.textContent = "บันทึกการกำกับดูแล";
          submitButton.disabled = false;
          document.getElementById('dataForm').reset();
          document.getElementById('imagePreview').innerHTML = '';
        } catch (error) {
          loadingOverlay.style.display = 'none';
          Swal.fire({
            icon: 'error',
            title: 'เกิดข้อผิดพลาด',
            text: 'ไม่สามารถบันทึกหรือส่งข้อมูลได้: ' + error.message
          });
          submitButton.textContent = "บันทึกการกำกับดูแล";
          submitButton.disabled = false;
        }
      };
      reader.readAsDataURL(fileInput);
    });

    function fetchPromoterNames() {
      fetch('https://script.google.com/macros/s/AKfycbyE8tvOVYAAR6WOiCVDEb2Fp1Yq3ebTAri9ZQYonpduMjQCFzod8xi0glBVXc6A6_qOeA/exec?action=getPromoterNames')
        .then(response => response.json())
        .then(data => {
          if (data.error) {
            Swal.fire({ icon: 'error', title: 'ข้อผิดพลาด', text: 'ไม่สามารถโหลดชื่อได้: ' + data.error });
            return;
          }
          const selectElement = $('#firstName');
          data.forEach(name => selectElement.append(new Option(name, name)));
          selectElement.select2({ placeholder: "กรุณาเลือกชื่อ", allowClear: true });
        })
        .catch(error => {
          Swal.fire({ icon: 'error', title: 'ข้อผิดพลาด', text: 'เกิดข้อผิดพลาดในการเชื่อมต่อ: ' + error.message });
        });
    }

    function fetchDistricts() {
      fetch('https://script.google.com/macros/s/AKfycbyE8tvOVYAAR6WOiCVDEb2Fp1Yq3ebTAri9ZQYonpduMjQCFzod8xi0glBVXc6A6_qOeA/exec?action=getDistricts')
        .then(response => response.json())
        .then(data => {
          if (data.error) {
            Swal.fire({ icon: 'error', title: 'ข้อผิดพลาด', text: 'ไม่สามารถโหลดเขตได้: ' + data.error });
            return;
          }
          const selectElement = $('#district');
          data.forEach(district => selectElement.append(new Option(district, district)));
          selectElement.select2({ placeholder: "กรุณาเลือกเขต", allowClear: true });

          // Add event listener for district change
          selectElement.on('change', function() {
            const selectedDistrict = $(this).val();
            if (selectedDistrict) {
              fetchAutoData(selectedDistrict);
            } else {
              document.getElementById('village').value = '';
              document.getElementById('subDistrict').value = '';
            }
          });
        })
        .catch(error => {
          Swal.fire({ icon: 'error', title: 'ข้อผิดพลาด', text: 'เกิดข้อผิดพลาดในการเชื่อมต่อ: ' + error.message });
        });
    }

    function fetchAutoData(district) {
      fetch(`https://script.google.com/macros/s/AKfycbyE8tvOVYAAR6WOiCVDEb2Fp1Yq3ebTAri9ZQYonpduMjQCFzod8xi0glBVXc6A6_qOeA/exec?action=getAutoData&district=${encodeURIComponent(district)}`)
        .then(response => response.json())
        .then(data => {
          if (data.error) {
            Swal.fire({ icon: 'error', title: 'ข้อผิดพลาด', text: 'ไม่สามารถโหลดข้อมูลได้: ' + data.error });
            return;
          }
          document.getElementById('village').value = data.village || '';
          document.getElementById('subDistrict').value = data.subDistrict || '';
        })
        .catch(error => {
          Swal.fire({ icon: 'error', title: 'ข้อผิดพลาด', text: 'เกิดข้อผิดพลาดในการเชื่อมต่อ: ' + error.message });
        });
    }

    window.onload = function() {
      initializeLiff().then(() => {
        getLocation();
        const today = new Date().toISOString().split('T')[0];
        document.getElementById('date').value = today;
        fetchPromoterNames();
        fetchDistricts();
      });
    };
  </script>
